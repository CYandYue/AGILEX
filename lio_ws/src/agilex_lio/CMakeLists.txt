cmake_minimum_required(VERSION 3.0.2)
project(agilex_lio)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs livox_ros_driver2)
catkin_package(CATKIN_DEPENDS roscpp sensor_msgs livox_ros_driver2)
include_directories(${catkin_INCLUDE_DIRS})
add_executable(sensor_relay src/sensor_relay.cpp)
target_compile_features(sensor_relay PRIVATE cxx_std_14)
target_link_libraries(sensor_relay ${catkin_LIBRARIES})
add_dependencies(sensor_relay ${catkin_EXPORTED_TARGETS})
install(TARGETS sensor_relay RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
catkin_install_python(PROGRAMS scripts/sensor_frames.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
